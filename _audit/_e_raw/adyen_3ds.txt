                keyPattern = keyPattern.replace( '+', '\\+' );
                keyPattern = keyPattern.replace( '?', '\\?' );

                // make sure the pattern starts with
                keyPattern = '^' + keyPattern;

                //make sure the pattern ends with
                keyPattern = keyPattern + '$';
                keyPattern = keyPattern.replace( '$$', '$' );

                // create new pattern with the given redirect key
                let pattern = new RegExp( keyPattern, "i");

                // replace the pattern with the corresponding redirect, in the current page
                let redirect = fullPath.replace( pattern , parsedRedirects[key] );

                // check if the final redirect url differs from the initial one
                if ( redirect !== fullPath )
                {
                    // Found a redirect match: currentPagePath => redirect
                    window.location.replace(redirect);

                    return ;
                }
                else if ( key.indexOf('$') < 0 )
                {
                    // if it does not match, lest check if we are looking at a sub-page redirect
                    keyPattern = keyPattern.replace( '$', '/(.*)' );

                    let subPattern = new RegExp( keyPattern, "i");

                    // replace the pattern with the corresponding redirect, in the current page
                    let subredirect = fullPath.replace( subPattern , parsedRedirects[key] + "/$1" );

                    if ( subredirect !== fullPath )
                    {
                        // Found a redirect match: currentPagePath => subredirect
                        window.location.replace(subredirect);

                        return ;
                    }
                }
            }
        }
    })();
